import csv
import platform
import statistics
import sys
from collections import Counter
from time import perf_counter

FILES = tuple(f"Module 3-1 Activity Data Set-{i}.csv" for i in range(1, 6))
HEADER = ["key", "data-a", "data-b", "validation"]
REPEATS = 3


def read_records(filename):
    """Read pipe-delimited records; preserve validation text and duplicates."""
    records = []
    with open(filename, newline="", encoding="utf-8-sig") as source:
        reader = csv.reader(source, delimiter="|")
        if next(reader, None) != HEADER:
            raise ValueError(f"{filename}: unexpected header")
        for line, row in enumerate(reader, start=2):
            if len(row) != 4:
                raise ValueError(f"{filename}, line {line}: expected four fields")
            try:
                key = int(row[0])
            except ValueError as error:
                raise ValueError(f"{filename}, line {line}: invalid key") from error
            if not row[3] or any(c not in "0123456789abcdefABCDEF" for c in row[3]):
                raise ValueError(f"{filename}, line {line}: invalid hexadecimal text")
            records.append((key, row[1], row[2], row[3]))
    return records


def sift_down(records, root, size):
    """Restore a max heap iteratively, comparing numeric keys only."""
    while 2 * root + 1 < size:
        child = 2 * root + 1
        if child + 1 < size and records[child + 1][0] > records[child][0]:
            child += 1
        if records[root][0] >= records[child][0]:
            return
        records[root], records[child] = records[child], records[root]
        root = child


def heap_sort(records):
    """Sort ascending in place; equal-key order is not guaranteed stable."""
    size = len(records)
    for root in range(size // 2 - 1, -1, -1):
        sift_down(records, root, size)
    for end in range(size - 1, 0, -1):
        records[0], records[end] = records[end], records[0]
        sift_down(records, 0, end)


def self_check():
    """Exercise boundaries, order patterns, and repeated keys."""
    for keys in ([], [1], [1, 2, 3], [3, 2, 1], [2, 1, 2], [7, 7, 7]):
        rows = [(key, str(i), "word", "a") for i, key in enumerate(keys)]
        before = Counter(rows)
        heap_sort(rows)
        assert [r[0] for r in rows] == sorted(keys)
        assert Counter(rows) == before


def main():
    self_check()
    print("CS 680 Module Three | Heap sort by numeric key")
    print(f"Python {platform.python_version()} | {platform.system()} {platform.machine()}")
    print(f"Boundary checks: PASS | {REPEATS} timed sorting trials per file")
    print("Times exclude copies, verification, and file writing. Load timed separately.")
    for filename in FILES:
        start = perf_counter()
        original = read_records(filename)
        load_time = perf_counter() - start
        before = Counter(original)
        timings = []
        for _ in range(REPEATS):
            records = original.copy()
            start = perf_counter()
            heap_sort(records)
            timings.append(perf_counter() - start)
            assert all(records[i - 1][0] <= records[i][0] for i in range(1, len(records)))
            assert Counter(records) == before
        output_name = filename.removesuffix(".csv") + "_sorted.csv"
        with open(output_name, "w", newline="", encoding="utf-8") as destination:
            writer = csv.writer(destination, delimiter="|")
            writer.writerow(HEADER)
            writer.writerows(records)
        duplicates = len(original) - len({row[0] for row in original})
        print(f"\n{filename}: {len(original):,} records; repeated keys={duplicates}")
        print(f"  Load {load_time:.6f}s | Sort median {statistics.median(timings):.6f}s")
        print("  Trials: " + ", ".join(f"{t:.6f}s" for t in timings))
        print("  Ascending order: PASS | Complete record preservation: PASS")
        if records:
            print(f"  Minimum key={records[0][0]} | Maximum key={records[-1][0]}")
        print(f"  Saved {output_name}")
    print("\nAll five data sets completed successfully.")


if __name__ == "__main__":
    try:
        main()
    except (OSError, ValueError, AssertionError) as error:
        print(f"ERROR: {error or 'verification failed'}", file=sys.stderr)
        sys.exit(1)
