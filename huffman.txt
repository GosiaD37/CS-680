"""Lossless byte-based Huffman codec for CS 680 Project Two.
Verify writes round-trip artifacts into a new destination directory.

import argparse
from collections import Counter
from dataclasses import dataclass
from hashlib import sha256
import heapq
from pathlib import Path
import struct
from time import perf_counter


# Magic, decoded size, payload bit count, decoded SHA-256.
HEADER = struct.Struct(">4sQQ32s")
FREQUENCIES = struct.Struct(">256Q")
MAGIC = b"HUF1"


@dataclass
class Node:
    symbol: int = -1
    left: object = None
    right: object = None


def build_tree(frequencies):
    """Build the same tree on both sides, including ties."""
    heap = [(count, symbol, Node(symbol))
            for symbol, count in enumerate(frequencies) if count]
    heapq.heapify(heap)
    next_id = 256
    while len(heap) > 1:
        left_count, _, left = heapq.heappop(heap)
        right_count, _, right = heapq.heappop(heap)
        heapq.heappush(heap, (left_count + right_count, next_id,
                              Node(left=left, right=right)))
        next_id += 1
    return heap[0][2] if heap else None


def build_codes(root):
    codes = {}

    def visit(node, bits, width):
        if node.symbol >= 0:
            codes[node.symbol] = (bits, max(1, width))
            return
        visit(node.left, bits << 1, width + 1)
        visit(node.right, (bits << 1) | 1, width + 1)

    if root is not None:
        visit(root, 0, 0)
    return codes


def encode(data):
    counts = Counter(data)
    frequencies = [counts.get(symbol, 0) for symbol in range(256)]
    codes = build_codes(build_tree(frequencies))
    payload = bytearray()
    pending = width = 0
    for symbol in data:
        code, size = codes[symbol]
        pending = (pending << size) | code
        width += size
        while width >= 8:
            width -= 8
            payload.append(pending >> width)
            pending &= (1 << width) - 1
    bit_count = 8 * len(payload) + width
    if width:
        payload.append(pending << (8 - width))
    return (HEADER.pack(MAGIC, len(data), bit_count, sha256(data).digest()),
            FREQUENCIES.pack(*frequencies), payload)


def decode(archive):
    offset = HEADER.size + FREQUENCIES.size
    if len(archive) < offset:
        raise ValueError("Incomplete archive header")
    magic, original_size, bit_count, digest = HEADER.unpack_from(archive)
    frequencies = FREQUENCIES.unpack_from(archive, HEADER.size)
    payload = memoryview(archive)[offset:]
    if (magic != MAGIC or sum(frequencies) != original_size
            or len(payload) != (bit_count + 7) // 8):
        raise ValueError("Invalid archive metadata or payload length")
    if bit_count % 8 and payload[-1] & ((1 << (8 - bit_count % 8)) - 1):
        raise ValueError("Invalid padding")
    root = build_tree(frequencies)
    codes = build_codes(root)
    expected_bits = sum(frequencies[symbol] * width
                        for symbol, (_, width) in codes.items())
    if bit_count != expected_bits:
        raise ValueError("Invalid payload bit count")

    if root is None:
        data = b""
    elif root.symbol >= 0:
        if any(payload):
            raise ValueError("Invalid single-symbol code")
        data = bytes([root.symbol]) * original_size
    else:
        data = bytearray()
        node = root
        full_bytes, tail = divmod(bit_count, 8)
        for position, octet in enumerate(payload):
            stop = 7 - tail if position == full_bytes else -1
            for shift in range(7, stop, -1):
                node = node.right if (octet >> shift) & 1 else node.left
                if node.symbol >= 0:
                    data.append(node.symbol)
                    node = root
        if node is not root:
            raise ValueError("Incomplete final code")
    if len(data) != original_size or sha256(data).digest() != digest:
        raise ValueError("Decoded data failed the length or SHA-256 check")
    return data


def write_archive(path, parts):
    with path.open("xb") as output:
        for part in parts:
            output.write(part)


def write_restored(path, data):
    with path.open("xb") as output:
        output.write(data)


def timed(label, operation, *args):
    start = perf_counter()
    result = operation(*args)
    print(f"{label:<24} {(perf_counter() - start) * 1000:10.3f} ms",
          flush=True)
    return result


def verify(source, destination):
    destination.mkdir()
    archive = destination / "compressed.huf"
    restored = destination / "restored.csv"
    print(f"Input: {source.name}", flush=True)
    original = timed("Read original", source.read_bytes)
    parts = timed("Encode", encode, original)
    timed("Write archive", write_archive, archive, parts)
    archive_size = archive.stat().st_size
    del parts
    packed = timed("Read archive", archive.read_bytes)
    data = timed("Decode", decode, packed)
    timed("Write restored", write_restored, restored, data)
    del packed, data
    if original != restored.read_bytes():
        raise ValueError("Restored file differs from source")
    print(f"Bytes: {len(original):,} -> {archive_size:,}")
    if original:
        print(f"Reduction: {100 * (1 - archive_size / len(original)):.2f}%")
    print(f"Byte comparison: PASS | SHA-256: {sha256(original).hexdigest()}")


def main():
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("mode", choices=("compress", "decompress", "verify"))
    parser.add_argument("source", type=Path)
    parser.add_argument("destination", type=Path)
    args = parser.parse_args()
    try:
        if args.mode == "compress":
            write_archive(args.destination, encode(args.source.read_bytes()))
        elif args.mode == "decompress":
            write_restored(args.destination, decode(args.source.read_bytes()))
        else:
            verify(args.source, args.destination)
    except (OSError, ValueError) as error:
        parser.exit(1, f"Error: {error}\n")


if __name__ == "__main__":
    main()
